cmake_minimum_required(VERSION 3.16)
project(ContactInEKF LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(contact_core STATIC src/backend.cpp)
target_include_directories(contact_core PUBLIC include)
target_link_libraries(contact_core PUBLIC Eigen3::Eigen)
add_executable(contact_inekf main.cpp)
target_link_libraries(contact_inekf PRIVATE contact_core)
